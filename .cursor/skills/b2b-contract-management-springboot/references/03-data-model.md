# 03 DB設計 / PostgreSQL / JPA
## テーブル
- parties(party_id PK, legal_name, corporate_number, created_at)
- contracts(contract_id PK, parent_contract_id FK self, contract_no UNIQUE, name, type, status, start_date, end_date, amount NUMERIC(18,2), currency_code VARCHAR(3), auto_renewal, notice_days, row_version, created_at, updated_at)
- contract_parties(contract_party_id PK, contract_id FK, party_id FK, party_role, UNIQUE(contract_id,party_id,party_role))
- contract_versions(version_id PK, contract_id FK, version_no, snapshot_json, created_at, UNIQUE(contract_id,version_no))
- contract_documents(document_id PK, contract_id FK, version_id FK, storage_key, sha256, content_type, uploaded_at)
- approval_requests(request_id PK, contract_id FK, version_id FK, status, requested_by, requested_at, idempotency_key UNIQUE)
- approval_steps(step_id PK, request_id FK, step_order, approver_id, status)  -- PENDING / WAITING / APPROVED / RETURNED / REJECTED / CANCELLED
- approval_histories(history_id PK, request_id FK, actor_id, action, comment_text, acted_at)
- contract_renewals(renewal_id PK, contract_id FK, new_end_date, decision, decided_at)
- payment_schedules(schedule_id PK, contract_id FK, due_date, amount NUMERIC(18,2), status)
- audit_logs(audit_id PK, actor_id, resource_type, resource_id, action, occurred_at, request_id)
## DDL例（Flyway: V1__create_contracts.sql）
```sql
CREATE TABLE contracts (
 contract_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
 parent_contract_id BIGINT REFERENCES contracts(contract_id),
 contract_no VARCHAR(50) NOT NULL UNIQUE,
 name VARCHAR(200) NOT NULL,
 type VARCHAR(50) NOT NULL,
 status VARCHAR(30) NOT NULL DEFAULT 'DRAFT',
 start_date DATE,
 end_date DATE,
 amount NUMERIC(18,2),
 currency_code VARCHAR(3) NOT NULL DEFAULT 'JPY',
 auto_renewal BOOLEAN NOT NULL DEFAULT FALSE,
 notice_days INTEGER,
 row_version BIGINT NOT NULL DEFAULT 0,
 created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
 updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
 CONSTRAINT chk_contract_period CHECK (end_date IS NULL OR start_date IS NULL OR end_date >= start_date),
 CONSTRAINT chk_notice_days CHECK (notice_days IS NULL OR notice_days >= 0)
);
CREATE INDEX idx_contracts_status ON contracts(status);
CREATE INDEX idx_contracts_end_date ON contracts(end_date);
CREATE INDEX idx_contracts_parent ON contracts(parent_contract_id);
```
JPAでは `@Version` をrowVersionに対応させる。FlywayのみがDDLを更新し、本番のHibernate `ddl-auto` はvalidate。`currency_code` は VARCHAR(3) に揃える（CHAR(3) だと Hibernate validate で型不一致になる）。監査の不変性・削除制限・親子循環防止も別途設計する。承認申請の対象は `approval_requests.version_id` であり、契約の最新版ではない。
